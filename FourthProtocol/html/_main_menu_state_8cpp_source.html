192void MainMenuState::handleEvent(const sf::Event& t_event)
193{
194 // Handle mouse movement for hover effects
195 if (const auto* mouseMove = t_event.getIf<sf::Event::MouseMoved>())
196 {
197 sf::Vector2f mousePos(static_cast<float>(mouseMove->position.x),
198 static_cast<float>(mouseMove->position.y));
199
200 // Check which button is being hovered
201 if (isMouseOver(m_vsAIButton, mousePos)) {
202 m_currentHover = ButtonHover::VsAI;
203 }
204 else if (isMouseOver(m_vsPvPButton, mousePos)) {
205 m_currentHover = ButtonHover::VsPvP;
206 }
207 else if (isMouseOver(m_hostGameButton, mousePos)) {
208 m_currentHover = ButtonHover::HostGame;
209 }
210 else if (isMouseOver(m_joinGameButton, mousePos)) {
211 m_currentHover = ButtonHover::JoinGame;
212 }
213 else if (isMouseOver(m_quitButton, mousePos)) {
214 m_currentHover = ButtonHover::Quit;
215 }
216 // Check what difficulty is being hovered over
217 else if (isMouseOverRectShape(m_easyButton, mousePos)) {
218 m_currentHover = ButtonHover::Easy;
219 }
220 else if (isMouseOverRectShape(m_mediumButton, mousePos)) {
221 m_currentHover = ButtonHover::Medium;
222 }
223 else if (isMouseOverRectShape(m_hardButton, mousePos)) {
224 m_currentHover = ButtonHover::Hard;
225 }
226 else {
227 m_currentHover = ButtonHover::None;
228 }
229 }
230
231 // Handle mouse clicks
232 if (const auto* mousePress = t_event.getIf<sf::Event::MouseButtonPressed>())
233 {
234 if (mousePress->button == sf::Mouse::Button::Left)
235 {
236 sf::Vector2f mousePos(static_cast<float>(mousePress->position.x),
237 static_cast<float>(mousePress->position.y));
238
239 if (isMouseOverRectShape(m_easyButton, mousePos)) {
240 m_selectedDifficulty = AIDifficulty::Easy;
241 std::cout << "Selected difficulty: Easy\n";
242 }
243 else if (isMouseOverRectShape(m_mediumButton, mousePos)) {
244 m_selectedDifficulty = AIDifficulty::Medium;
245 std::cout << "Selected difficulty: Medium\n";
246 }
247 else if (isMouseOverRectShape(m_hardButton, mousePos)) {
248 m_selectedDifficulty = AIDifficulty::Hard;
249 std::cout << "Selected difficulty: Hard\n";
250 }
251
252 else if (isMouseOver(m_vsAIButton, mousePos))
253 {
254 std::cout << "Starting game vs AI\n";
255 m_nextState = std::make_unique<GamePlayState>(m_sharedFont, GameMode::VsAI, m_selectedDifficulty);
256 m_transitionRequested = true;
257 }
258 else if (isMouseOver(m_vsPvPButton, mousePos))
259 {
260 std::cout << "Starting game vs Player\n";
261 m_nextState = std::make_unique<GamePlayState>(m_sharedFont, GameMode::VsPvP);
262 m_transitionRequested = true;
263 }
264 else if (isMouseOver(m_hostGameButton, mousePos))
265 {
266 std::cout << "Hosting network game\n";
267
268 // Non-blocking start
269 if (m_networkManager->startHostAsync())
270 {
271 m_nextState = std::make_unique<LobbyState>(m_sharedFont, m_networkManager, true);
272 m_transitionRequested = true;
273 }
274 else
275 {
276 std::cout << "Failed to start host!\n";
277 }
278 }
279 else if (isMouseOver(m_joinGameButton, mousePos))
280 {
281 std::cout << "\n=== Join Network Game ===\n";
282 std::cout << "Enter host IP address: ";
283
284 std::string l_ipInput;
285 std::cin >> l_ipInput;
286
287 std::cout << "Connecting to " << l_ipInput << "...\n";
288
289 // Non-blocking connect
290 if (m_networkManager->connectAsClientAsync(l_ipInput))
291 {
292 m_nextState = std::make_unique<LobbyState>(m_sharedFont, m_networkManager, false);
293 m_transitionRequested = true;
294 }
295 else
296 {
297 std::cout << "Failed to connect! Check IP and try again.\n";
298 std::cout << "Press any key to continue...\n";
299 std::cin.ignore();
300 std::cin.get();
301 }
302 }
303
304 else if (isMouseOver(m_quitButton, mousePos))
305 {
306 std::cout << "Quit button clicked\n";
307 m_nextState = nullptr;
308 m_transitionRequested = true;
309 }
310 }

336sf::Color MainMenuState::getDifficultyButtonColor(AIDifficulty t_difficulty, bool t_isSelected, bool t_isHovered) const
337{
338 // Base colors for each difficulty
339 sf::Color baseColor;
340 switch (t_difficulty)
341 {
342 case AIDifficulty::Easy: baseColor = sf::Color(100.f, 255.f, 150.f); break;
343 case AIDifficulty::Medium: baseColor = sf::Color(200.f, 200.f, 50.f); break;
344 case AIDifficulty::Hard: baseColor = sf::Color(255.f, 100.f, 100.f); break;
345 }
346
347 // If selected, make brighter
348 if (t_isSelected)
349 {
350 return baseColor;
351 }
352
353 // If hovered (but not selected), make slightly brighter
354 if (t_isHovered)
355 {
356 return sf::Color(
357 std::min(255, baseColor.r + 50),
358 std::min(255, baseColor.g + 50),
359 std::min(255, baseColor.b + 50)
360 );
361 }
362
363 // Otherwise, make darker
364 return sf::Color(baseColor.r / 2, baseColor.g / 2, baseColor.b / 2);
365}
