49void Game::run()
50{
51 sf::Clock clock;
52 sf::Time timeSinceLastUpdate = sf::Time::Zero;
53 const float fps{ 60.0f };
54 sf::Time timePerFrame = sf::seconds(1.0f / fps); // 60 fps
55 while (m_window.isOpen())
56 {
57 processEvents(); // as many as possible
58 timeSinceLastUpdate += clock.restart();
59 while (timeSinceLastUpdate > timePerFrame)
60 {
61 timeSinceLastUpdate -= timePerFrame;
62 processEvents(); // at least 60 fps
63 update(timePerFrame); //60 fps
64 }
65 render(); // as many as possible
66 }
67}
